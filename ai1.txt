2. GENETIC ALGORITHM FOR N-QUEENS (N=6)
============================================================

CODE:

import random

n = int(input("Enter N: "))
population_size = int(input("Population size: "))
generations = int(input("Maximum generations: "))

def fitness(c):
    return sum(c[i] != c[j] and abs(c[i] - c[j]) != j-i
               for i in range(n) for j in range(i+1, n))

population = [random.sample(range(n), n)
              for _ in range(population_size)]

for generation in range(generations):
    population.sort(key=fitness, reverse=True)
    best = population[0]

    if fitness(best) == n*(n-1)//2:
        print("Solution:", best)
        print("Chromosome encoding: one queen per column; values are rows")
        break

    new_population = population[:2]
    while len(new_population) < population_size:
        p1, p2 = random.choices(population[:max(2, population_size//2)], k=2)
        cut = random.randint(1, n-1)
        child = p1[:cut] + [x for x in p2 if x not in p1[:cut]]
        if random.random() < 0.3:
            i, j = random.sample(range(n), 2)
            child[i], child[j] = child[j], child[i]
        new_population.append(child)
    population = new_population
else:
    print("No solution found within generation limit")

3. A* SEARCH FOR THE 8-PUZZLE
============================================================

CODE:

import heapq

start = tuple(map(int, input("Enter 9 numbers (use 0 for blank): ").split()))
goal = tuple(map(int, input("Enter goal state: ").split()))

def h(s):
    return sum(abs(i//3 - (v-1)//3) + abs(i%3 - (v-1)%3)
               for i, v in enumerate(s) if v)

def neighbors(s):
    i = s.index(0)
    r, c = divmod(i, 3)
    for dr, dc in ((1,0),(-1,0),(0,1),(0,-1)):
        nr, nc = r+dr, c+dc
        if 0 <= nr < 3 and 0 <= nc < 3:
            j = nr*3+nc
            t = list(s)
            t[i], t[j] = t[j], t[i]
            yield tuple(t)

pq = [(h(start), 0, start, [])]
best = {start: 0}

while pq:
    f, g, s, path = heapq.heappop(pq)
    if s == goal:
        for state in path + [s]:
            print(state[:3], state[3:6], state[6:])
        print("Moves:", g)
        break
    if g != best[s]:
        continue
    for t in neighbors(s):
        ng = g+1
        if ng < best.get(t, float("inf")):
            best[t] = ng
            heapq.heappush(pq, (ng+h(t), ng, t, path+[s]))
else:
    print("No solution")

SAMPLE INPUT:

1 2 3 4 5 6 7 8 0

SAMPLE OUTPUT:

(1, 2, 3) (4, 5, 6) (7, 0, 8)
(1, 2, 3) (4, 5, 6) (7, 8, 0)
Moves: 2

HEURISTICS:
h1 = number of misplaced tiles.
h2 = sum of Manhattan distances.
Manhattan distance usually guides A* better while remaining admissible.

    u, v, w = map(int, input("u v cost: ").split())

    if node == goal:

        print("Cost:", cost)

    if cost != best[node]:
        continue
    for nxt, w in graph[node]:
        nc = cost+w
        if nc < best.get(nxt,float("inf")):
            best[nxt] = nc
            heapq.heappush(pq,(nc,nxt,path+[nxt]))
else:

6. SIMULATED ANNEALING
============================================================

CODE:

import math
import random

x = float(input("Initial x: "))
temperature = float(input("Initial temperature: "))
cooling = float(input("Cooling factor (0 to 1): "))
steps = int(input("Iterations: "))

def objective(x):
    return x*x + 10*math.sin(x)

best = x
while temperature > 0.001 and steps > 0:
    candidate = x + random.uniform(-1,1)
    delta = objective(candidate)-objective(x)
    if delta < 0 or random.random() < math.exp(-delta/temperature):
        x = candidate
    if objective(x) < objective(best):
        best = x
    temperature *= cooling
    steps -= 1

print("Best x:", best)
print("Minimum value:", objective(best))

15. ALPHA-BETA PRUNING
============================================================

CODE:

import math

def alphabeta(tree, depth, alpha, beta, maximizing):
    if not isinstance(tree, list):
        return tree
    if maximizing:
        value = -math.inf
        for child in tree:
            value = max(value, alphabeta(child, depth+1, alpha, beta, False))
            alpha = max(alpha, value)
            if alpha >= beta:
                break
        return value
    value = math.inf
    for child in tree:
        value = min(value, alphabeta(child, depth+1, alpha, beta, True))
        beta = min(beta, value)
        if beta <= alpha:
            break
    return value

tree = eval(input("Enter nested game tree, e.g. [[3,5],[2,9]]: "))
print("Best value:", alphabeta(tree, 0, -math.inf, math.inf, True))

18. PROLOG: CNF AND RESOLUTION
============================================================

KNOWLEDGE BASE:

% P -> Q is equivalent to not(P) OR Q.
% Q -> R is equivalent to not(Q) OR R.
% P is true.

p.
q :- p.
r :- q.

QUERY:

?- r.

SAMPLE OUTPUT:
true.

CNF CLAUSES:
1. not(P) OR Q
2. not(Q) OR R
3. P

RESOLUTION:
From 1 and 3, derive Q.
From 2 and Q, derive R.

Therefore R is true.

import math

board = list(input("Enter 9 cells using X, O, -: ").upper())
player = input("AI player (X/O): ").upper()
human = "O" if player == "X" else "X"
wins = [(0,1,2),(3,4,5),(6,7,8),(0,3,6),
        (1,4,7),(2,5,8),(0,4,8),(2,4,6)]

def winner(b):
    for a,c,d in wins:
        if b[a] == b[c] == b[d] and b[a] != "-":
            return b[a]
    return None

def minimax(b, turn):
    w = winner(b)
    if w == player: return 1
    if w == human: return -1
    if "-" not in b: return 0
    scores = []
    for i in range(9):
        if b[i] == "-":
            b[i] = turn
            scores.append((minimax(b, human if turn == player else player), i))
            b[i] = "-"
    return (max if turn == player else min)(scores)[0]

best_move = None
best_score = -math.inf
for i in range(9):
    if board[i] == "-":
        board[i] = player
        score = minimax(board, human)
        board[i] = "-"
        if score > best_score:
            best_score, best_move = score, i

if best_move is not None:
    board[best_move] = player
print(" ".join(board[i:i+3] for i in range(0,9,3)))

grid = []
for _ in range(9):
    grid.append(list(map(int, input("Enter 9 digits (0=empty): ").split())))

def safe(r,c,v):
    if v in grid[r]: return False
    if any(grid[i][c] == v for i in range(9)): return False
    br, bc = (r//3)*3, (c//3)*3
    return all(grid[i][j] != v
               for i in range(br,br+3) for j in range(bc,bc+3))

def solve():
    for r in range(9):
        for c in range(9):
            if grid[r][c] == 0:
                for v in range(1,10):
                    if safe(r,c,v):
                        grid[r][c] = v
                        if solve(): return True
                        grid[r][c] = 0
                return False
    return True

if solve():
    for row in grid: print(*row)
else:
    print("No solution")

from collections import deque

n = int(input("Number of nodes: "))
m = int(input("Number of edges: "))
graph = {i: [] for i in range(n)}

for _ in range(m):
    u,v = map(int, input("Edge u v: ").split())
    graph[u].append(v)
    graph[v].append(u)

start = int(input("Start node: "))

visited = {start}
q = deque([start])
bfs = []

while q:
    u = q.popleft()
    bfs.append(u)
    for v in graph[u]:
        if v not in visited:
            visited.add(v)
            q.append(v)

visited = set()
dfs = []

def visit(u):
    visited.add(u)
    dfs.append(u)
    for v in graph[u]:
        if v not in visited:
            visit(v)

visit(start)
print("BFS:", bfs)
print("DFS:", dfs)

23. WATER JUG PROBLEM USING DFS

from collections import deque

a = int(input("Capacity of jug A: "))
b = int(input("Capacity of jug B: "))
target = int(input("Target amount: "))

stack = [(0,0,[])]
visited = set()
found = False

while stack:
    x,y,path = stack.pop()
    if (x,y) in visited: continue
    visited.add((x,y))
    path = path + [(x,y)]
    if x == target or y == target:
        print("Path:", path)
        found = True
        break
    states = [
        (a,y), (x,b), (0,y), (x,0),
        (x-min(x,b-y), y+min(x,b-y)),
        (x+min(y,a-x), y-min(y,a-x))
    ]
    for s in states:
        if s not in visited:
            stack.append((s[0],s[1],path))

if not found:
    print("No solution")

rows = int(input("Rows: "))

            ng=g+1
            if ng<best.get((nr,nc),float("inf")):
                best[(nr,nc)]=ng
                heapq.heappush(pq,(ng+h(nr,nc),ng,nr,nc,path+[(nr,nc)]))

(Covers: SUN+FUN=SWIM, SEND+MORE=MONEY, TWO+TWO=FOUR)
Constraints: all letters different digits, no leading zero, every lower-column sum must already be correct (mod 10^k).

puzzle = input("Enter puzzle (e.g. SEND + MORE = MONEY): ").upper().replace(" ", "")
left, right = puzzle.split("=")
words = left.split("+") + [right]
sign = [1] * (len(words) - 1) + [-1]
L = max(map(len, words))
letters, cnt = [], [0]
for k in range(L):                      # order letters column by column (right to left)
    for w in words:
        if k < len(w) and w[-1 - k] not in letters:
            letters.append(w[-1 - k])
    cnt.append(len(letters))
lead = {w[0] for w in words}
W = [{} for _ in range(L + 1)]          # W[k][letter] = weight using only the lowest k columns
for k in range(L + 1):
    for w, s in zip(words, sign):
        for p in range(min(k, len(w))):
            W[k][w[-1 - p]] = W[k].get(w[-1 - p], 0) + s * 10 ** p
sols = []
def solve(i, used, a):
    if i == len(letters):
        if sum(W[L][c] * a[c] for c in a) == 0: sols.append(dict(a))
        return
    for d in range(10):
        if d in used or (d == 0 and letters[i] in lead): continue
        a[letters[i]] = d
        # constraint: lowest k columns must already add up correctly (mod 10^k)
        if all(sum(W[k][c] * a[c] for c in W[k]) % 10 ** k == 0 for k in range(1, L) if cnt[k] == i + 1):
            solve(i + 1, used | {d}, a)
        del a[letters[i]]
solve(0, set(), {})
if not sols: print("No solution")
for s in sols:
    print("Letters:", s)
    print(" + ".join(str(int("".join(str(s[c]) for c in w))) for w in words[:-1]), "=", int("".join(str(s[c]) for c in words[-1])))

2. GENETIC ALGORITHM (N-Queens, N=6 or any N)
==============================================================================
(Covers: genetic algorithm; GA for N-Queens with chromosome encoding, crossover, mutation)
Chromosome = list of N numbers, c[i] = row of the queen in column i. Fitness = non-attacking pairs. Selection = tournament, crossover = single point, mutation = random gene change, elitism = 2.

import random
n = int(input("N (board size): "))
pop = int(input("Population size: "))
mr = float(input("Mutation rate (0-1): "))
gens = int(input("Max generations: "))
best = n * (n - 1) // 2
def fit(c):  # chromosome c[i] = row of queen in column i ; fitness = non-attacking pairs
    return best - sum(c[i] == c[j] or abs(c[i] - c[j]) == j - i for i in range(n) for j in range(i + 1, n))
P = [[random.randrange(n) for _ in range(n)] for _ in range(pop)]
for g in range(gens):
    P.sort(key=fit, reverse=True)
    if g % 10 == 0 or fit(P[0]) == best: print("Gen", g, "best", P[0], "fitness", fit(P[0]), "/", best)
    if fit(P[0]) == best: break
    new = P[:2]                                            # elitism
    while len(new) < pop:
        a = max(random.sample(P, 3), key=fit); b = max(random.sample(P, 3), key=fit)  # tournament selection
        cp = random.randint(1, n - 1); child = a[:cp] + b[cp:]                         # single-point crossover
        if random.random() < mr: child[random.randrange(n)] = random.randrange(n)       # mutation
        new.append(child)
    P = new
print("Solution chromosome:", P[0], "fitness", fit(P[0]))
for r in range(n): print(" ".join("Q" if P[0][c] == r else "." for c in range(n)))

    a = [x for x in t if x]
    return sum(a[i] > a[j] for i in range(len(a)) for j in range(i + 1, len(a))) + (t.index(0) // n if n % 2 == 0 else 0)

(Covers: Wumpus world pit and gold)
Symbols P(r,c)=pit. KB: visited cell has no pit; Breeze(x) <=> OR(pit in adjacent cells). Entailment checked by enumerating all models.
Glitter in a visited cell means gold is there.

vis = {}
for _ in range(k):
    r, c, b, g = map(int, input("row col breeze(0/1) glitter(0/1): ").split())
    vis[(r, c)] = (b, g)
cells = [(r, c) for r in range(1, N + 1) for c in range(1, N + 1)]
idx = {x: i for i, x in enumerate(cells)}
def adj(r, c): return [(a, b) for a, b in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)) if (a, b) in idx]
# KB: visited cell has no pit ; Breeze(x) <=> OR(Pit in neighbours of x)
def kb(m): return all(not m[idx[x]] and bool(b) == any(m[idx[y]] for y in adj(*x)) for x, (b, g) in vis.items())

if not models: print("KB inconsistent"); raise SystemExit

gold = [x for x, (b, g) in vis.items() if g]
if gold: print("\nGold found at:", gold)
else:

6. SIMULATED ANNEALING (travelling salesman)
==============================================================================
(Covers: simulated annealing to reach the goal state)

import random, math
n = int(input("Number of cities: "))

while T > Tmin:

    new = cur[:i] + cur[i:j + 1][::-1] + cur[j + 1:]       # 2-opt neighbour
    delta = d(new) - d(cur)
    if delta < 0 or random.random() < math.exp(-delta / T): cur = new
    if d(cur) < d(best): best = cur[:]
    T *= cool
print("Best tour:", best + [best[0]], "\nLength:", round(d(best), 3))

class GaussianNB:                      # Naive Bayes implemented from scratch
    def fit(s, X, y):
        s.c = np.unique(y)
        s.m = np.array([X[y == k].mean(0) for k in s.c])
        s.v = np.array([X[y == k].var(0) for k in s.c]) + 1e-9 * X.var(0).max()

        return s.c[np.argmax(ll, axis=0)]

11. WATER JUG AS A CONSTRAINT SATISFACTION PROBLEM (CSP)
==============================================================================
(Covers: CSP formulation of water jug with defined constraints and goal)
Variables S0..Sk (state after each step). Domain = (a,b) with 0<=a<=capA, 0<=b<=capB. Constraints: S0=(0,0), each S(i+1) is a legal move from S(i), no repeated state, target in S(k).

a = int(input("Capacity of jug A: ")); b = int(input("Capacity of jug B: ")); t = int(input("Target amount (in either jug): "))
dom = [(x, y) for x in range(a + 1) for y in range(b + 1)]            # domain of every variable S_i
def ok(s, n):                                                          # transition constraint: n must follow from s by a legal pour/fill/empty
    p = min(s[0], b - s[1]); q = min(s[1], a - s[0])
    return n in [(a, s[1]), (s[0], b), (0, s[1]), (s[0], 0), (s[0] - p, s[1] + p), (s[0] + q, s[1] - q)] and n != s
def solve(path, k):                                                    # backtracking over S_0 .. S_k
    if len(path) == k + 1: return path if t in path[-1] else None      # goal constraint on S_k
    for n in dom:
        if n not in path and ok(path[-1], n):                          # all-different + transition
            r = solve(path + [n], k)
            if r: return r
for k in range(1, len(dom)):                                           # iterative deepening on number of steps
    r = solve([(0, 0)], k)                                             # initial constraint S_0 = (0,0)
    if r: print("Variables S0..S%d =" % k, r); break
else: print("No solution")

import re
from itertools import product
def parse(s):                                   # grammar:  <=>  lowest, then =>, |, &, ~   (use parentheses)
    t = re.findall(r"<=>|=>|[A-Za-z_]\w*|[~&|()]", s); p = [0]
    def eat(): p[0] += 1; return t[p[0] - 1]
    def peek(): return t[p[0]] if p[0] < len(t) else None
    def iff():
        l = imp()
        while peek() == "<=>": eat(); r = imp(); l = (lambda a, b: lambda m: a(m) == b(m))(l, r)
        return l
    def imp():
        l = orr()
        if peek() == "=>": eat(); r = imp(); return (lambda a, b: lambda m: (not a(m)) or b(m))(l, r)
        return l
    def orr():
        l = andd()
        while peek() == "|": eat(); r = andd(); l = (lambda a, b: lambda m: a(m) or b(m))(l, r)
        return l
    def andd():
        l = neg()
        while peek() == "&": eat(); r = neg(); l = (lambda a, b: lambda m: a(m) and b(m))(l, r)
        return l
    def neg():
        x = eat()
        if x == "~": f = neg(); return lambda m: not f(m)
        if x == "(": f = iff(); eat(); return f
        return lambda m: m[x]
    return iff(), set(re.findall(r"[A-Za-z_]\w*", s))

17. CART-POLE: Q-LEARNING AGENT (gymnasium)
==============================================================================
(Covers: cart pole implementation and train the agent)

import gymnasium as gym, numpy as np
episodes = int(input("Training episodes: ")); alpha = float(input("Learning rate: ")); gamma = float(input("Discount factor: "))
env = gym.make("CartPole-v1")
bins = (1, 1, 6, 12); lo = np.array([-2.4, -3, -0.21, -3.5]); hi = -lo
def disc(o):
    r = np.clip((o - lo) / (hi - lo), 0, 0.9999); return tuple((r * bins).astype(int))
Q = np.zeros(bins + (2,)); eps = 1.0
for ep in range(episodes):
    o, _ = env.reset(); s = disc(o); done = False; tot = 0
    while not done:
        a = env.action_space.sample() if np.random.rand() < eps else int(np.argmax(Q[s]))
        o, r, term, trunc, _ = env.step(a); s2 = disc(o); done = term or trunc; tot += r
        Q[s + (a,)] += alpha * (r + gamma * (0 if term else np.max(Q[s2])) - Q[s + (a,)]); s = s2
    eps = max(0.01, eps * 0.995)
    if (ep + 1) % max(1, episodes // 10) == 0: print(f"Episode {ep + 1}: reward {tot:.0f}  epsilon {eps:.3f}")
res = []
for _ in range(10):                                     # evaluate greedy policy
    o, _ = env.reset(); done = False; tot = 0
    while not done:
        o, r, term, trunc, _ = env.step(int(np.argmax(Q[disc(o)]))); done = term or trunc; tot += r
    res.append(tot)

18. SUDOKU SOLVER (CSP: backtracking + most-constrained-variable)
==============================================================================
(Covers: constraint satisfaction problem - sudoku solver)

print("Enter 9 rows of 9 digits (use 0 or . for blank):")
g = [[0 if ch in "0." else int(ch) for ch in input().strip()] for _ in range(9)]
def cand(r, c):   # domain = digits not used in row, column, 3x3 box (constraints)
    used = set(g[r]) | {g[i][c] for i in range(9)} | {g[r//3*3+i][c//3*3+j] for i in range(3) for j in range(3)}
    return [d for d in range(1, 10) if d not in used]
def solve():      # backtracking + MRV (most constrained variable first)
    e = [(len(cand(r, c)), r, c) for r in range(9) for c in range(9) if g[r][c] == 0]
    if not e: return True
    _, r, c = min(e)
    for d in cand(r, c):
        g[r][c] = d
        if solve(): return True
        g[r][c] = 0
    return False
if solve():
    print("\nSolution:")
    for r in g: print(" ".join(map(str, r)))
else: print("No solution")

from collections import deque
e = int(input("Number of edges: ")); t = {}
print("Enter edges as: parent child")
for _ in range(e):
    p, c = input().split(); t.setdefault(p, []).append(c)
root = input("Root node: "); goal = input("Goal node: ")
def bfs():
    q = deque([[root]]); order = []
    while q:
        p = q.popleft(); order.append(p[-1])
        if p[-1] == goal: return order, p
        q.extend(p + [c] for c in t.get(p[-1], []))
    return order, None
def dfs():
    st = [[root]]; order = []
    while st:
        p = st.pop(); order.append(p[-1])
        if p[-1] == goal: return order, p
        st.extend(p + [c] for c in reversed(t.get(p[-1], [])))
    return order, None
for name, f in (("BFS", bfs), ("DFS", dfs)):
    o, p = f(); print(f"{name} visit order: {' '.join(o)}\n{name} path to goal: {' -> '.join(p) if p else 'not found'}")

20. A* ALGORITHM FOR GRID MAP
==============================================================================
(Covers: A* algorithm for grid map)
S start, G goal, # wall, . free. Moves: 4 directions, cost 1, heuristic = Manhattan distance. * marks the path.

import heapq
r, c = map(int, input("Rows Cols: ").split())
print("Enter grid rows (S=start, G=goal, #=wall, .=free):")
g = [list(input().strip()) for _ in range(r)]
find = lambda ch: next((i, j) for i in range(r) for j in range(c) if g[i][j] == ch)
s, goal = find("S"), find("G")
h = lambda p: abs(p[0] - goal[0]) + abs(p[1] - goal[1])             # Manhattan heuristic
pq = [(h(s), 0, s)]; cost = {s: 0}; par = {s: None}; done = set()
while pq:
    f, gc, u = heapq.heappop(pq)
    if u in done: continue
    done.add(u)
    if u == goal: break
    for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
        v = (u[0] + dr, u[1] + dc)
        if 0 <= v[0] < r and 0 <= v[1] < c and g[v[0]][v[1]] != "#" and gc + 1 < cost.get(v, 1e9):
            cost[v] = gc + 1; par[v] = u; heapq.heappush(pq, (gc + 1 + h(v), gc + 1, v))
if goal not in par: print("No path")
else:
    p, u = [], goal
    while u: p.append(u); u = par[u]
    for i, j in p[1:-1]: g[i][j] = "*"
    print("Path length:", len(p) - 1, "\nPath:", p[::-1]); print("\n".join("".join(x) for x in g))
