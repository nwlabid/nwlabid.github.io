import heapq

n = int(input("enter no of nodes: "))
graph = []
for _ in range(n):
    graph.append(list(map(int, input().split())))

print("enter heuristic values:")
h = list(map(int, input().split()))

start, goal = map(int, input("enter start and end node: ").split())
pq = [(h[start], 0, start, [start])]
visited = set()

while pq:
    f, g, node, path = heapq.heappop(pq)
    
    if node == goal:
        print("path: " + "->".join(map(str, path)))
        print("cost:", g)
        break
        
    if node in visited:
        continue
    visited.add(node)
    
    for i in range(n):
        if graph[node][i] != 0 and i not in visited:
            new_g = g + graph[node][i]
            new_f = new_g + h[i]
            heapq.heappush(pq, (new_f, new_g, i, path + [i]))
else:
    print("No path found")

import heapq

n = int(input("Enter number of nodes: "))
print("Enter the cost adjacency matrix (0 for no edge):")
graph = []
for i in range(n):
    graph.append(list(map(int, input().split())))

start = int(input("Enter start node: "))
goal = int(input("Enter goal node: "))
pq = [(0, start, [start])]
visited = [False] * n

while pq:
    cost, node, path = heapq.heappop(pq)
    
    if visited[node]:
        continue
    visited[node] = True
    
    if node == goal:
        print("\nGoal Found!")
        print("Path:", " -> ".join(map(str, path)))
        print("Cost:", cost)
        break
        
    for i in range(n):
        if graph[node][i] > 0 and not visited[i]:
            heapq.heappush(pq, (cost + graph[node][i], i, path + [i]))
else:
    print("Goal not found!")

simulated annealing.py


import random
import math

n = int(input())
graph = []
for i in range(n):
    graph.append(list(map(int, input().split())))

def cost(path):
    s = 0
    for i in range(n - 1):
        s += graph[path[i]][path[i + 1]]
    s += graph[path[-1]][path[0]]
    return s

path = list(range(n))
best = path[:]
T = 1000

while T > 1:
    new = path[:]
    i = random.randint(0, n - 1)
    j = random.randint(0, n - 1)
    new[i], new[j] = new[j], new[i]
    d = cost(new) - cost(path)
    if d < 0 or random.random() < math.exp(-d / T):
        path = new[:]
    if cost(path) < cost(best):
        best = path[:]
    T *= 0.95

print("Best Tour:")
print(*(best, best[0]))
print("Minimum Cost =", cost(best))

import random
import math

def f(x):
    return x*x - 4*x + 4

x = random.randint(-10, 10)
best = x
T = 1000

while T > 1:
    new = x + random.uniform(-1, 1)
    d = f(new) - f(x)

    if d < 0 or random.random() < math.exp(-d / T):
        x = new

    if f(x) < f(best):
        best = x

    T *= 0.95

print("Best x =", best)

waterjug.py


cap_a = int(input("Enter capacity of Jug A: "))
cap_b = int(input("Enter capacity of Jug B: "))
target = int(input("Enter target amount: "))
queue = [((0, 0), [])]
visited = {(0, 0)}

while queue:
    (a, b), path = queue.pop(0)
    if a == target or b == target:
        print("\nSolution Found!")
        for step in path:
            print(step)
        print("Final State:", (a, b))
        break
        
    states = [
        ((cap_a, b), "Fill A"),
        ((a, cap_b), "Fill B"),
        ((0, b), "Empty A"),
        ((a, 0), "Empty B"),
        (
            (max(0, a - (cap_b - b)), min(cap_b, a + b)),
            "Pour A -> B"
        ),
        (
            (min(cap_a, a + b), max(0, b - (cap_a - a))),
            "Pour B -> A"
        )
    ]
    
    for new_state, action in states:
        if new_state not in visited:
            visited.add(new_state)
            queue.append(
                (new_state, path + [f"{action} -> {new_state}"])
            )
else:
    print("No solution exists.")

def solve(a, b, c):
    letters = set(a + b + c)
    letters = list(letters)
    
    def backtrack(i, used, mp):
        if i == len(letters):
            x = int(''.join(str(mp[ch]) for ch in a))
            y = int(''.join(str(mp[ch]) for ch in b))
            z = int(''.join(str(mp[ch]) for ch in c))
            if x + y == z:
                print(mp)
                print(x, "+", y, "=", z)
                return True
            return False
            
        ch = letters[i]
        for digit in range(10):
            if digit not in used:
                mp[ch] = digit
                if backtrack(i + 1, used | {digit}, mp):
                    return True
                del mp[ch]
        return False
        
    backtrack(0, set(), {})

a = input("First word: ")
b = input("Second word: ")
c = input("Result word: ")

import random

def binary_to_decimal(s):
    return int(s, 2)

def get_x(s, low, high):
    d = binary_to_decimal(s)
    return low + (high - low) * d / (2 ** len(s) - 1)

n = int(input("Population size: "))
length = int(input("Chromosome length: "))
seed = int(input("Seed: "))
low, high = map(int, input("Min Max: ").split())
generations = int(input("Generations: "))

random.seed(seed)
pop = [format(random.randrange(2 ** length), f'0{length}b') for _ in range(n)]

for g in range(generations):
    x = [get_x(i, low, high) for i in pop]
    fit = [i * i for i in x]
    print(f"\nGeneration", g + 1)
    for i in range(n):
        print(pop[i], "x =", round(x[i], 2), "fitness =", round(fit[i], 2))
        
    best = pop[fit.index(max(fit))]
    new_pop = []
    
    for _ in range(n // 2):
        p1 = random.choices(pop, weights=fit)[0]
        p2 = random.choices(pop, weights=fit)[0]
        point = random.randint(1, length - 1)
        c1 = p1[:point] + p2[point:]
        c2 = p2[:point] + p1[point:]
        
        def mutate(c):
            c = list(c)
            for i in range(length):
                if random.random() < 0.01:
                    c[i] = '1' if c[i] == '0' else '0'
            return "".join(c)
            
        new_pop.append(mutate(c1))
        new_pop.append(mutate(c2))
        
    new_pop[0] = best
    pop = new_pop[:n]

x = [get_x(i, low, high) for i in pop]
fit = [i * i for i in x]
best = fit.index(max(fit))
print("\nFinal Best")
print("Chromosome:", pop[best])
print("x:", x[best])
print("Fitness:", fit[best])

import gymnasium as gym
import numpy as np
env = gym.make("CartPole-v1")
bins = [3, 3, 12, 12]
bounds = [(-2.4, 2.4),(-0.5, 0.5),(-0.2095, 0.2095),(-0.87, 0.87),]
q_table = np.zeros((*bins, 2))
alpha = 0.15
gamma = 0.99
epsilon = 1.0
epsilon_min = 0.01
epsilon_decay = 0.998
episodes = 3000
def discretize_state(state):
    discrete = []
    for i in range(4):
        value = np.clip(state[i], bounds[i][0], bounds[i][1])
        ratio = (value - bounds[i][0]) / (bounds[i][1] - bounds[i][0])
        index = int(ratio * bins[i])
        index = min(index, bins[i] - 1)
        discrete.append(index)
    return tuple(discrete)
for episode in range(episodes):
    state, info = env.reset()
    state = discretize_state(state)
    total_reward = 0
    for step in range(500):
        if np.random.random() < epsilon:
            action = env.action_space.sample()
        else:
            action = int(np.argmax(q_table[state]))
        next_state, reward, terminated, truncated, info = env.step(action)
        next_state = discretize_state(next_state)
        old_value = q_table[state + (action,)]
        next_max = np.max(q_table[next_state])
        target = reward + gamma * next_max * (not terminated)
        q_table[state + (action,)] = old_value + alpha * (target - old_value)
        state = next_state
        total_reward += reward
        if terminated or truncated:
            break
    epsilon = max(epsilon_min, epsilon * epsilon_decay)
    if (episode + 1) % 100 == 0:
        print(
            f"Episode {episode + 1}, "
            f"Reward: {total_reward}, "
            f"Epsilon: {epsilon:.3f}"
        )
env.close()
test_env = gym.make("CartPole-v1", render_mode="human")
for episode in range(3):
    state, info = test_env.reset()
    state = discretize_state(state)
    total_reward = 0
    while True:
        action = int(np.argmax(q_table[state]))
        next_state, reward, terminated, truncated, info = test_env.step(action)
        state = discretize_state(next_state)
        total_reward += reward
        if terminated or truncated:
            break
    print(f"Test episode {episode + 1}: {total_reward} steps")
test_env.close()
